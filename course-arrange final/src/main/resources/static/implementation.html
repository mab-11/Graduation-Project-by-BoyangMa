4.1 CHROMOSOME ENCODING
The chromosome encoding scheme represents a complete course schedule. Each gene contains a pair of values: timeslot ID and room ID for each teaching task. 
// Individual.java - Chromosome Encoding
public class Individual {
    private int[] chromosome;  // [task1_time, task1_room, task2_time, task2_room, ...]
    private double fitness;
    private int hardConstraintPenalty;
    private double softConstraintScore;

    // Constructor: Initialize chromosome with random genes
    public Individual(int geneLength) {
        this.chromosome = new int[geneLength * 2];
        for (int i = 0; i < geneLength; i++) {
            chromosome[i*2] = randomTimeslot();     // Random timeslot
            chromosome[i*2+1] = randomRoom();        // Random room
        }
    }
}
4.2 FITNESS CALCULATION
The fitness function evaluates solution quality based on three components: hard constraints (mandatory), soft constraints (optimization goals), and resource utilization. 
// FitnessCalculator.java - Multi-objective Fitness
public class FitnessCalculator {

    // Main fitness calculation
    public double calculateFitness(Individual indiv) {
        int hardPenalty = calcHardConstraintPenalty(indiv);
        double softScore = calcSoftConstraintScore(indiv);
        double resourceUtil = calcResourceUtilization(indiv);

        // Hard constraints first: penalty=0 means all hard constraints satisfied
        if (hardPenalty == 0) {
            return 100 + softScore + resourceUtil * 10;
        } else {
            return 100.0 / (hardPenalty + 1);
        }
    }

    // Hard constraint penalty calculation
    private int calcHardConstraintPenalty(Individual indiv) {
        int penalty = 0;
        penalty += classTimeClashes * 10;    // Class time conflict
        penalty += teacherTimeClashes * 10;  // Teacher time conflict
        penalty += roomTimeClashes * 10;     // Room time conflict
        penalty += capacityViolations * 5;   // Room capacity insufficient
        return penalty;
    }

    // Soft constraint score calculation
    private double calcSoftConstraintScore(Individual indiv) {
        double score = 0;
        score += teacherPreferenceScore * 15;   // Teacher preference
        score += courseConcentration * 10;       // Course concentration
        score += roomBalance * 5;               // Room balance
        score += dayConcentration * 8;          // Day concentration
        return score;
    }
}
4.3 GENETIC OPERATORS
The genetic algorithm employs three main operators: selection (tournament selection), crossover (uniform crossover), and mutation (adaptive mutation). 
// GA.java - Genetic Operators

// Tournament Selection
public Individual tournamentSelection(Population pop, int tournamentSize) {
    Population tournament = new Population(tournamentSize);
    for (int i = 0; i < tournamentSize; i++) {
        int randomIdx = random.nextInt(pop.size());
        tournament.setIndividual(i, pop.getIndividual(randomIdx));
    }
    double bestFitness = -1;
    Individual best = null;
    for (int i = 0; i < tournament.size(); i++) {
        if (tournament.getIndividual(i).getFitness() > bestFitness) {
            bestFitness = tournament.getIndividual(i).getFitness();
            best = tournament.getIndividual(i);
        }
    }
    return best;
}

// Uniform Crossover
public Individual crossover(Individual parent1, Individual parent2) {
    Individual child = new Individual(parent1.getChromosomeLength());
    for (int i = 0; i < child.getChromosomeLength(); i++) {
        if (random.nextDouble() < crossoverRate) {
            child.setGene(i, parent1.getGene(i));
        } else {
            child.setGene(i, parent2.getGene(i));
        }
    }
    return child;
}

// Adaptive Mutation
public void mutate(Individual indiv) {
    double currentMutationRate = calcAdaptiveMutationRate(indiv);
    for (int i = 0; i < indiv.getChromosomeLength(); i++) {
        if (random.nextDouble() < currentMutationRate) {
            indiv.setGene(i*2, randomTimeslot());
            indiv.setGene(i*2+1, randomRoom());
        }
    }
}
4.4 ADAPTIVE PARAMETER CONTROL
The adaptive mechanism dynamically adjusts mutation rate based on population diversity. When premature convergence is detected, the mutation rate is increased to explore new solutions. 
// AdaptiveGA.java - Adaptive Parameter Control
public class AdaptiveGA extends GA {

    private double baseMutationRate = 0.05;
    private double minMutationRate = 0.001;
    private double maxMutationRate = 0.3;
    private double convergenceThreshold = 0.001;

    // Calculate adaptive mutation rate based on population diversity
    public double calcAdaptiveMutationRate(Individual indiv) {
        double fitnessVariance = calcPopulationFitnessVariance();

        // Premature convergence detected - increase mutation
        if (fitnessVariance < convergenceThreshold) {
            mutationRate = Math.min(maxMutationRate, mutationRate * 1.5);
        } else {
            // Normal evolution - gradually decrease mutation
            mutationRate = Math.max(minMutationRate, mutationRate * 0.9);
        }
        return mutationRate;
    }

    // Calculate population diversity
    private double calcPopulationFitnessVariance() {
        double mean = calcPopulationMeanFitness();
        double variance = 0;
        for (Individual indiv : population.getIndividuals()) {
            variance += Math.pow(indiv.getFitness() - mean, 2);
        }
        return variance / population.size();
    }
}
4.5 MAIN ALGORITHM LOOP
The main evolution loop performs selection, crossover, and mutation iteratively until convergence or maximum generations are reached. 
// GA.java - Main Evolution Loop
public Individual evolve(Population population) {
    Population newPopulation = new Population(population.size());

    // Elitism: Keep best individuals
    for (int i = 0; i < eliteCount; i++) {
        newPopulation.setIndividual(i, population.getFittest(i));
    }

    // Evolution through selection, crossover, mutation
    for (int i = eliteCount; i < population.size(); i++) {
        Individual parent1 = tournamentSelection(population, tournamentSize);
        Individual parent2 = tournamentSelection(population, tournamentSize);

        Individual child;
        if (random.nextDouble() < crossoverRate) {
            child = crossover(parent1, parent2);
        } else {
            child = new Individual(parent1);
        }

        mutate(child);
        newPopulation.setIndividual(i, child);
    }

    return newPopulation.getFittest(0);
}
